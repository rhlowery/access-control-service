23      @Test
24      public void testUserServiceBranches() {
25          String userId = "test-service-user";
26          // clear branch
27          userService.clear();
28          
29          // saveUser branch
30          User user = new User(userId, "Test Service User", "test@example.com", "STANDARD_USER", List.of(), "ADMIN");
31          userService.saveUser(user);
32          
33          // getUser branch found
34          Optional<User> found = userService.getUser(userId);
35          assertTrue(found.isPresent());
36          
37          // listUsers branch
38          List<User> users = userService.listUsers();
39          assertFalse(users.isEmpty());
40          
41          // updateUserPersona branch
42          userService.updateUserPersona(userId, "DATA_SCIENTIST");
43          found = userService.getUser(userId);
44          assertEquals("DATA_SCIENTIST", found.get().persona());
45          
46          // updateUserGroups branch
47          userService.updateUserGroups(userId, List.of("group1"));
48          found = userService.getUser(userId);
49          assertTrue(found.get().groups().contains("group1"));
50      }

52      @Test
53      public void testAccessRequestServiceBranches() {
54          String requestId = "req-service-1";
55          String userId = "user-service-1";
56          
57          // clear branch
58          accessRequestService.clear();
59          
60          // saveRequests NEW branch
61          AccessRequest req = new AccessRequest(requestId, userId, userId, "USER", "cat", "sch", "tab", "TABLE", List.of("READ"), "PENDING", 0L, 0L, "test", null, List.of("approver-group"), null, null);
62          accessRequestService.saveRequests(List.of(req), userId, List.of(), false);
63          
64          // getRequestById branch found
65          Optional<AccessRequest> found = accessRequestService.getRequestById(requestId);
66          assertTrue(found.isPresent());
67          
68          // getAllRequests branches
69          // isAdmin branch
70          List<AccessRequest> all = accessRequestService.getAllRequests(userId, List.of(), true);
71          assertFalse(all.isEmpty());
72          
73          // isOwner branch
74          all = accessRequestService.getAllRequests(userId, List.of(), false);
75          assertFalse(all.isEmpty());
76          
77          // isApprover branch
78          all = accessRequestService.getAllRequests("other-user", List.of("approver-group"), false);
79          assertFalse(all.isEmpty());
80          
81          // saveRequests UPDATE branch (as owner)
82          AccessRequest update = new AccessRequest(requestId, userId, userId, "USER", "cat", "sch", "tab", "TABLE", List.of("READ"), "APPROVED", 0L, 0L, "update", null, List.of("approver-group"), null, null);
83          accessRequestService.saveRequests(List.of(update), userId, List.of(), false);
84          
85          // saveRequests UPDATE branch (as admin)
86          accessRequestService.saveRequests(List.of(update), "admin", List.of(), true);
87          
88          // saveRequests FORBIDDEN branch
89          assertThrows(RuntimeException.class, () -> accessRequestService.saveRequests(List.of(update), "intruder", List.of(), false));
90      }
